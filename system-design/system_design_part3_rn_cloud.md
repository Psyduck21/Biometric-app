### 10.3 Package-Based Platform Abstraction

> **No custom Kotlin/Swift written.** The following community packages ship their own
> native bindings internally. You only write TypeScript.

#### Package Selection

| Concern | Package | Notes |
|---|---|---|
| TFLite inference | `@tensorflow/tfjs` + `@tensorflow/tfjs-react-native` | JS runtime; delegates to GPU via WebGL/Metal |
| Face camera + frame processing | `react-native-vision-camera` v4 | Frame Processors via JSI (JS, no Kotlin/Swift) |
| Keystore / Secure Enclave | `react-native-keychain` | Abstracts Android Keystore + iOS Keychain/Enclave |
| AES encryption | `react-native-aes-crypto` | AES-256-GCM in JS |
| Root / Jailbreak detection | `react-native-jail-monkey` | JS API, detects both platforms |
| Device info / fingerprint | `react-native-device-info` | UniqueId, hardware info |
| SQLite (encrypted) | `op-sqlite` with SQLCipher flag | Or `expo-sqlite` for simpler setup |
| Location / GPS | `expo-location` | Pure RN, no native code |
| Network status | `@react-native-community/netinfo` | Connectivity detection |
| Crypto (hashing) | `expo-crypto` | SHA-256, HMAC |

```typescript
// src/services/ai/EmbeddingService.ts
import * as tf from '@tensorflow/tfjs';
import '@tensorflow/tfjs-react-native';  // registers RN backend
import { fetch as tfFetch, bundleResourceIO } from '@tensorflow/tfjs-react-native';

export class EmbeddingService {
  private model: tf.GraphModel | null = null;

  async initialize(): Promise<void> {
    await tf.ready(); // wait for RN backend (uses GPU via Metal/WebGL)
    // Load bundled TFLite-converted model (SavedModel format for tfjs)
    const modelJson = require('../../../assets/models/mobilefacenet/model.json');
    const modelWeights = require('../../../assets/models/mobilefacenet/weights.bin');
    this.model = await tf.loadGraphModel(bundleResourceIO(modelJson, modelWeights));
  }

  async generateEmbedding(alignedFace: Uint8Array): Promise<Float32Array> {
    if (!this.model) throw new Error('Model not initialized');

    const tensor = tf.tidy(() => {
      const img = tf.tensor3d(alignedFace, [112, 112, 3], 'float32')
        .div(127.5)
        .sub(1.0)             // normalize to [-1, 1]
        .expandDims(0);       // [1, 112, 112, 3]
      return this.model!.predict(img) as tf.Tensor;
    });

    const embedding = await tensor.data() as Float32Array;
    tensor.dispose();
    return embedding;         // 512-dim vector
  }

  computeCosineSimilarity(a: Float32Array, b: Float32Array): number {
    let dot = 0, normA = 0, normB = 0;
    for (let i = 0; i < a.length; i++) {
      dot   += a[i] * b[i];
      normA += a[i] * a[i];
      normB += b[i] * b[i];
    }
    return dot / (Math.sqrt(normA) * Math.sqrt(normB));
  }
}
```

```typescript
// src/services/security/CryptoService.ts
import * as Keychain from 'react-native-keychain'; // Keystore + Secure Enclave
import AesCrypto from 'react-native-aes-crypto';
import * as Crypto from 'expo-crypto';

export class CryptoService {
  private static SERVICE_KEY = 'com.biometric.app.masterkey';

  // Stores master key securely (Android Keystore / iOS Keychain + Secure Enclave)
  async storeMasterKey(key: string): Promise<void> {
    await Keychain.setGenericPassword('masterkey', key, {
      service: CryptoService.SERVICE_KEY,
      accessible: Keychain.ACCESSIBLE.WHEN_UNLOCKED_THIS_DEVICE_ONLY,
      securityLevel: Keychain.SECURITY_LEVEL.SECURE_HARDWARE, // enforces Keystore/Enclave
    });
  }

  async getMasterKey(): Promise<string> {
    const creds = await Keychain.getGenericPassword({ service: CryptoService.SERVICE_KEY });
    if (!creds) throw new Error('Master key not found');
    return creds.password;
  }

  async encrypt(plaintext: string, key: string): Promise<EncryptedPayload> {
    const iv = await AesCrypto.randomKey(16);  // 128-bit IV
    const ciphertext = await AesCrypto.encrypt(plaintext, key, iv, 'aes-256-cbc');
    return { ciphertext, iv };
  }

  async decrypt(payload: EncryptedPayload, key: string): Promise<string> {
    return AesCrypto.decrypt(payload.ciphertext, key, payload.iv, 'aes-256-cbc');
  }

  async sha256(data: string): Promise<string> {
    return Crypto.digestStringAsync(Crypto.CryptoDigestAlgorithm.SHA256, data);
  }

  async generateNonce(): Promise<string> {
    return Crypto.getRandomBytesAsync(16).then(b =>
      Array.from(b).map(x => x.toString(16).padStart(2, '0')).join('')
    );
  }
}

interface EncryptedPayload { ciphertext: string; iv: string; }
```

```typescript
// src/services/security/SecurityCheckService.ts
import JailMonkey from 'react-native-jail-monkey';  // no Kotlin/Swift needed
import DeviceInfo from 'react-native-device-info';

export class SecurityCheckService {
  async isDeviceCompromised(): Promise<{ safe: boolean; reason?: string }> {
    if (JailMonkey.isJailBroken()) {
      return { safe: false, reason: 'root_or_jailbreak_detected' };
    }
    if (JailMonkey.canMockLocation()) {
      return { safe: false, reason: 'mock_location_detected' };
    }
    if (JailMonkey.isOnExternalStorage()) {
      return { safe: false, reason: 'running_from_external_storage' };
    }
    return { safe: true };
  }

  async getDeviceFingerprint(): Promise<string> {
    const uniqueId  = await DeviceInfo.getUniqueId();
    const brand     = DeviceInfo.getBrand();
    const model     = DeviceInfo.getModel();
    const systemVer = DeviceInfo.getSystemVersion();
    // Hash to produce stable device ID
    const raw = `${uniqueId}::${brand}::${model}::${systemVer}`;
    const { CryptoService } = await import('./CryptoService');
    return new CryptoService().sha256(raw);
  }
}
```

```typescript
import { useFrameProcessor } from 'react-native-vision-camera';

// Frame processor runs on JS thread via JSI — no Swift/Kotlin needed
export function useLivenessFrameProcessor(

) {
  const frameProcessor = useFrameProcessor((frame) => {
    'worklet';
```
